function doGet(e) {
  var action = e && e.parameter && e.parameter.action ? e.parameter.action : "";
  var result = {};

  if (action === "getDataInit") {
    result = getDataInit();
  } else if (action === "saveDiemDanhTheoSoLuong") {
    result = saveDiemDanhTheoSoLuong(e.parameter.ngay, e.parameter.buoi, e.parameter.soVang, e.parameter.dsVangStr);
  } else if (action === "saveDetailedAttendance") {
    result = saveDetailedAttendance(e.parameter.ngay, e.parameter.buoi, e.parameter.soVang, e.parameter.dsVangStr);
  } else if (action === "getAdminHistoryData") {
    result = getAdminHistoryData();
  } else {
    return HtmlService.createHtmlOutputFromFile('Index')
        .setTitle('Điểm Danh Sĩ Số Lớp 10A4')
        .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }

  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// Lấy dữ liệu khởi tạo ban đầu
function getDataInit() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("DanhSachHS");
  
  var danhSachHS = [];
  if (sheet) {
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0]) {
        danhSachHS.push({
          maHS: String(data[i][0]),
          hoTen: String(data[i][1]),
          ngaySinh: data[i][2] ? Utilities.formatDate(new Date(data[i][2]), Session.getScriptTimeZone(), "dd/MM/yyyy") : ""
        });
      }
    }
  }
  
  return {
    totalHS: danhSachHS.length > 0 ? danhSachHS.length : 42,
    danhSachHS: danhSachHS
  };
}

// Lưu điểm danh từ Mục 2 (Theo số lượng chung)
function saveDiemDanhTheoSoLuong(ngay, buoi, soVang, dsVangStr) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("ThongKe");
  
  if (!sheet) {
    sheet = ss.insertSheet("ThongKe");
    sheet.appendRow(["Ngày", "Buổi", "Tổng vắng", "Danh sách HS vắng", "Thời gian lưu"]);
  }
  
  var data = sheet.getDataRange().getValues();
  var rowIndex = -1;
  var targetNgay = String(ngay).trim();
  var targetBuoi = String(buoi).trim();
  
  for (var i = 1; i < data.length; i++) {
    var rowNgay = data[i][0];
    if (rowNgay instanceof Date) {
      rowNgay = Utilities.formatDate(rowNgay, Session.getScriptTimeZone(), "dd/MM/yyyy");
    } else {
      rowNgay = String(rowNgay).trim();
    }
    var rowBuoi = String(data[i][1]).trim();
    
    if (rowNgay === targetNgay && rowBuoi === targetBuoi) {
      rowIndex = i + 1;
      break;
    }
  }
  
  var currentTime = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss");
  
  if (rowIndex > 0) {
    sheet.getRange(rowIndex, 3).setValue(soVang);
    sheet.getRange(rowIndex, 4).setValue(dsVangStr);
    sheet.getRange(rowIndex, 5).setValue(currentTime);
  } else {
    sheet.appendRow([ngay, buoi, soVang, dsVangStr, currentTime]);
  }
  
  return {
    message: "Đã lưu điểm danh theo số lượng thành công cho ngày " + ngay + " (" + buoi + ")!",
    thoiGianLuu: currentTime
  };
}

// Lấy lịch sử điểm danh hiển thị cho giáo viên
function getAdminHistoryData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("ThongKe");
  if (!sheet) return [];
  
  var data = sheet.getDataRange().getValues();
  var history = [];
  
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) {
      var ngayStr = data[i][0];
      if (ngayStr instanceof Date) {
        ngayStr = Utilities.formatDate(ngayStr, Session.getScriptTimeZone(), "dd/MM/yyyy");
      }
      
      var thoiGianLuu = "Chưa có thông tin";
      if (data[i].length >= 5 && data[i][4]) {
        thoiGianLuu = String(data[i][4]);
      }
      
      history.push({
        ngay: String(ngayStr),
        buoi: String(data[i][1]),
        tongVang: data[i][2],
        ghiChuVang: data[i][3],
        thoiGianLuu: thoiGianLuu
      });
    }
  }
  return history;
}
